\begin{frame}{先让听众知道问题}
  \begin{block}{这一页只回答一件事}
    怎样同时交代结果、单位和适用条件？
  \end{block}
  \begin{itemize}
    \item 标题写出问题或主要信息。
    \item 图表只保留与解释有关的内容。
    \item 在页脚或图注中标明数据来源。
  \end{itemize}
\end{frame}
\begin{frame}{用两栏建立对应关系}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.47\textwidth}
      \textbf{左栏：说明}
      \begin{itemize}
        \item 右栏金额全部为虚构示例。
        \item 明确单位，避免读者猜测。
      \end{itemize}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{tabular}{lr}
        \toprule 项目 & 金额（元）\\\midrule
        甲 & 120\\ 乙 & 80\\ 合计 & 200\\\bottomrule
      \end{tabular}
    \end{column}
  \end{columns}
\end{frame}
