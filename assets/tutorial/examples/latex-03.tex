\section{逐步计算}
设 \(x\in\mathbb{R}\)，展开得到
\begin{align}
  (x+1)^2 &= x^2+2x+1,\label{eq:expand}\\
  (x+1)^2-x^2 &= 2x+1.\label{eq:difference}
\end{align}
式~\eqref{eq:difference}~由式~\eqref{eq:expand}~两边减去
\(x^2\) 得到。

\section{一个编号的多行公式}
\begin{equation}
\begin{split}
  (a+b)^2-(a-b)^2
    &= (a^2+2ab+b^2)\\
    &\quad -(a^2-2ab+b^2)\\
    &= 4ab.
\end{split}
\end{equation}

\section{分段定义}
\[
|x|=\begin{cases}
  x,  & x\geq 0,\\
  -x, & x<0.
\end{cases}
\]
